cmake_minimum_required(VERSION 3.22)
project(Wave723 VERSION 0.1.0 LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_POLICY_VERSION_MINIMUM 3.10)

include(FetchContent)
if(POLICY CMP0135)
    cmake_policy(SET CMP0135 NEW)
endif()
set(BUILD_SHARED_LIBS OFF)
set(UNICORN_ARCH arm CACHE STRING "Unicorn guest architectures")
set(UNICORN_BUILD_TESTS OFF)
set(UNICORN_INSTALL OFF)
set(UNICORN_LEGACY_STATIC_ARCHIVE OFF)
set(BUILD_EXAMPLES OFF)
set(INSTALL_PROJECT OFF)

FetchContent_Declare(unicorn
    URL https://codeload.github.com/unicorn-engine/unicorn/tar.gz/8028ec436f2d9376525352dd38ed9ed6b9f6be10)
FetchContent_Declare(miniz
    URL https://codeload.github.com/richgel999/miniz/tar.gz/293d4db1b7d0ffee9756d035b9ac6f7431ef8492)
FetchContent_MakeAvailable(unicorn miniz)

add_library(wave723_core STATIC src/firmware.cpp src/machine.cpp)
target_include_directories(wave723_core PUBLIC include)
target_link_libraries(wave723_core PRIVATE unicorn miniz)
if(MSVC)
    target_compile_options(wave723_core PRIVATE /W4)
else()
    target_compile_options(wave723_core PRIVATE -Wall -Wextra -Wpedantic)
endif()

if(ANDROID)
    add_library(wave723_android SHARED android/app/src/main/cpp/native.cpp)
    target_link_libraries(wave723_android PRIVATE wave723_core)
    return()
endif()

add_executable(wave723 src/cli.cpp)
target_link_libraries(wave723 PRIVATE wave723_core)
install(TARGETS wave723 RUNTIME DESTINATION .)
if(MINGW)
    target_link_options(wave723 PRIVATE -municode)
endif()

option(WAVE723_DESKTOP "Build the SDL desktop application" ON)
if(WAVE723_DESKTOP)
    set(SDL_SHARED OFF)
    set(SDL_STATIC ON)
    set(SDL_TEST_LIBRARY OFF)
    set(SDL_INSTALL OFF)
    FetchContent_Declare(SDL3
        URL https://codeload.github.com/libsdl-org/SDL/tar.gz/badbf8da4ee72b3ef599c721ffc9899e8d7c8d90)
    FetchContent_MakeAvailable(SDL3)
    add_executable(wave723-ui src/desktop.cpp)
    target_link_libraries(wave723-ui PRIVATE wave723_core SDL3::SDL3)
    install(TARGETS wave723-ui RUNTIME DESTINATION .)
endif()

include(CTest)
if(BUILD_TESTING)
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    add_test(NAME firmware-and-reset
        COMMAND "${Python3_EXECUTABLE}" "${PROJECT_SOURCE_DIR}/tests/smoke.py"
                "$<TARGET_FILE:wave723>")
    if(TARGET wave723-ui)
        add_test(NAME desktop-starts COMMAND wave723-ui --smoke)
        set_tests_properties(desktop-starts PROPERTIES ENVIRONMENT "SDL_VIDEODRIVER=dummy")
    endif()
endif()

install(FILES README.md LICENSE DESTINATION .)
install(FILES "${unicorn_SOURCE_DIR}/COPYING" DESTINATION licenses RENAME unicorn-COPYING)
install(FILES "${miniz_SOURCE_DIR}/LICENSE" DESTINATION licenses RENAME miniz-LICENSE)
if(WAVE723_DESKTOP)
    install(FILES "${sdl3_SOURCE_DIR}/LICENSE.txt" DESTINATION licenses RENAME SDL-LICENSE.txt)
endif()
